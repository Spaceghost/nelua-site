Pattern and playback speed

Changing the pattern starts a fresh board. Change speed at any time.

One tick through the toolchain

Imba controls playback and draws the grid. A Nelua WebAssembly function counts the neighbors. For each cell it calls a JavaScript function compiled by Bend to decide whether the cell lives.

The survival rule starts as typed Nelua inside a file-local bendjs block. The extension reuses the fork’s Bend2 lowering pass. Open the artifacts below to follow the same rule through the build.

#syntax 'syntax.bendjs'
## pragmas.noentrypoint = true
## pragmas.nogc = true
## pragmas.nochecks = true
## pragmas.nocstaticassert = true

-- The host supplies this rule as a Bend-generated JavaScript function.
bendjs (survives) do
  local function survives(alive: uint32, neighbors: uint32): uint32
    if neighbors == 3_u32 then
      return 1_u32
    elseif neighbors == 2_u32 then
      return alive
    else
      return 0_u32
    end
  end
end

-- The guest computes one row from three 16-bit rows on a wrapping board.
local function next_row(above: uint32, row: uint32, below: uint32): uint32
    <cexport, codename 'next_row'>
  local next_bits: uint32 = 0
  for x: uint32 = 0, < 16 do
    local left = (x + 15) % 16
    local right = (x + 1) % 16
    local count: uint32 = 0
    for offset: uint32 = 0, < 3 do
      local at = (left + offset) % 16
      count = count + ((above >> at) & 1) + ((below >> at) & 1)
    end
    count = count + ((row >> left) & 1) + ((row >> right) & 1)
    next_bits = next_bits | (survives((row >> x) & 1, count) << x)
  end
  return next_bits
end

function word_to_u32(w) {
  let x = 0;
  for (let i = 0; w.$ === "WCon"; i++) {
    x |= Number(w.head) << i;
    w = w.tail;
  }
  return x >>> 0;
}

function u32_to_word(x) {
  let w = {$: "WNil"};
  for (let i = 31; i >= 0; i--) {
    w = {$: "WCon", head: ((x >>> i) & 1) === 1, tail: w};
  }
  return w;
}

function f32_read(s) {
  const re = /^[\t\n\v\f\r ]*[+-]?((\d+(?:\.\d*)?|\.\d+)(e[+-]?\d+)?|inf(inity)?|nan)$/i;
  return re.test(s) ? {$: "Some", value: f32_round(
    s.replace(/inf\w*/i, "Infinity"))} : {$: "None"};
}

const f32_round = function f32_round(s) {
  const d = Number(s);
  const a = Math.abs(d);
  const f = Math.fround(a);
  const g = 2 * a - Math.min(f, 2 ** 128);
  if (g === f || Math.fround(g) !== g || g === Infinity) {
    return Math.sign(d) * f;
  }
  let k = 0;
  while (a * 2 ** k % 1 !== 0) {
    k += 1;
  }
  const [, i, r, e] = /(\d*)\.?(\d*)(?:e([+-]?\d+))?$/i.exec(s);
  const n = Number(e ?? 0) - r.length;
  const x = BigInt(i + r) * 2n ** BigInt(k) * 10n ** BigInt(Math.max(n, 0));
  const y = BigInt(a * 2 ** k) * 10n ** BigInt(Math.max(-n, 0));
  return Math.sign(d) * (x === y || x > y !== g > f ? f : g);
};

const char_new = function char_new(c) {
  return c < 55296 || c >= 57344 && c <= 1114111 ? String.fromCodePoint(c) : String.fromCharCode(56320, c >>> 22, c >>> 11 & 2047, c & 2047);
};

function char_len(s, at = 0) {
  return s[at] === "\uDC00" ? 4 : s.codePointAt(at) > 0xFFFF ? 2 : 1;
}

function char_code(s) {
  return s.length < 4 ? s.codePointAt(0)
    : (s.charCodeAt(1) << 22 | s.charCodeAt(2) << 11 | s.charCodeAt(3)) >>> 0;
}

function string_contains(s, p) {
  // A match starts at a Char, not inside a packed U32 code.
  for (let at = 0; at <= s.length; at += char_len(s, at)) {
    if (s.startsWith(p, at)) {
      return true;
    }
  }
  return false;
}

// Run
// ===

function run_tail(f, x) {
  return {$: "$JMP", f: f.j?.f === f ? f.j : f, x};
}

function run_clo(j) {
  const f = (x) => run_loop(j(x));
  f.j = j;
  j.f = f;
  return f;
}

function run_loop(r) {
  while (r !== null && typeof r === "object" && r.$ === "$JMP") {
    r = r.f(r.x);
  }
  return r;
}

function run_lib(f, n) {
  return (...a) => a.length < n ? run_lib((...b) => f(...a, ...b), n - a.length)
    : run_loop(f(...a));
}

function $nelua_lower_branch5(_test_0, _nelua_lower_value2_0, _nelua_lower_value3_0) {
  if (_test_0) {
    return _nelua_lower_value2_0;
  } else {
    return ((0 + 0) >>> 0);
  }
}

function $nelua_lower_branch4(_test_0, _nelua_lower_value2_0, _nelua_lower_value3_0) {
  if (_test_0) {
    return ((0 + 1) >>> 0);
  } else {
    const _x_0 = ((0 + 2) >>> 0);
    return $nelua_lower_branch5((_nelua_lower_value3_0 === _x_0), _nelua_lower_value2_0, _nelua_lower_value3_0);
  }
}

function $nelua_lower_function1(_nelua_lower_value2_0, _nelua_lower_value3_0) {
  const _x_0 = ((0 + 3) >>> 0);
  return $nelua_lower_branch4((_nelua_lower_value3_0 === _x_0), _nelua_lower_value2_0, _nelua_lower_value3_0);
}

function $survives(_x1_0, _x2_0) {
  return $nelua_lower_function1(_x1_0, _x2_0);
}

uint32_t nelua_shr_nluint32(uint32_t a, int32_t b) {
  if(NELUA_LIKELY(b >= 0 && b < 32)) {
    return (uint32_t)a >> b;
  } else if(NELUA_UNLIKELY(b < 0 && b > -32)) {
    return (uint32_t)a << -b;

Imba interface
const board = document.querySelector('#life-board')
const step = document.querySelector('#life-step')
const play = document.querySelector('#life-play')
const pattern = document.querySelector('#life-pattern')
const speed = document.querySelector('#life-speed')
const counter = document.querySelector('#life-generation')
let timer = null
const reset = document.querySelector('#life-reset')
const status = document.querySelector('#life-status')
const context = board.getContext('2d')
let generation = 0
let rows = new Uint16Array(16)

def draw
	context.fillStyle = '#172339'
	context.fillRect(0, 0, 320, 320)
	for y in [0...16]
		for x in [0...16]
			context.fillStyle = (rows[y] & (1 << x)) ? '#a5e2cc' : '#283b54'
			context.fillRect(x * 20 + 1, y * 20 + 1, 18, 18)
	board.setAttribute('aria-label', "Game of Life {pattern.value} at generation {generation}")
	counter.textContent = "Generation {generation}"

def pause
	if timer != null
		clearInterval(timer)
	timer = null
	play.textContent = 'Play'
	status.textContent = 'Paused'

def seed
	pause!
	rows.fill(0)
	if pattern.value == 'blinker'
		rows[7] = 7 << 6
	elif pattern.value == 'beacon'
		rows[6] = rows[7] = 3 << 5
		rows[8] = rows[9] = 3 << 7
	elif pattern.value == 'block'
		rows[7] = rows[8] = 3 << 7
	else
		rows[4] = 1 << 6
		rows[5] = 1 << 7
		rows[6] = 7 << 5
	generation = 0
	draw!

def load
	try
		const ruleURL = new URL('rule.mjs', document.baseURI).href
		const rules = await import(ruleURL)
		const response = await fetch('life.wasm')
		throw new Error("HTTP {response.status}") unless response.ok
		const survives = do(alive, neighbors) rules.default.survives(alive >>> 0, neighbors >>> 0)
		const wasm = await WebAssembly.instantiate(await response.arrayBuffer!, {env: {survives: survives}})
		def advance
			const next = new Uint16Array(16)
			for y in [0...16]
				next[y] = wasm.instance.exports.next_row(rows[(y + 15) % 16], rows[y], rows[(y + 1) % 16])
			rows = next
			generation += 1
			draw!
		def start
			clearInterval(timer)
			timer = setInterval(advance, Number(speed.value))
			play.textContent = 'Pause'
			status.textContent = 'Playing'
		speed.addEventListener('change') do
			start! if timer != null
		speed.disabled = false
		step.addEventListener('click') do
			pause!
			advance!
		play.addEventListener('click') do
			if timer == null
				start!
			else
				pause!
		document.addEventListener('visibilitychange') do
			pause! if document.hidden
		pattern.addEventListener('change', seed)
		play.disabled = false
		pattern.disabled = false
		reset.addEventListener('click', seed)
		reset.disabled = false
		step.disabled = false
		seed!
	catch error
		status.textContent = "Could not load the simulation: {error.message}"

load!
