## pragmas.noentrypoint = true
## pragmas.nogc = true
## pragmas.nochecks = true
## pragmas.nocstaticassert = true

-- Opcode declarations from Drift, c970832, lib/drift/asm.nelua (MIT).
local asm: type = @record{}

## local names = {'HALT','LIT','MOV','ADD','SUB','MUL','DIV','AND','OR',
##                'XOR','SHL','SHR','LD8','ST8','LD32','ST32','JMP',
##                'JEQ','JNE','JLT','IO','EVENT','SLEEP'}
## for i, name in ipairs(names) do
local asm.#|name|#: uint8 <comptime> = #[i-1]#
## end

local codes: [23]uint8 = {
  asm.HALT, asm.LIT, asm.MOV, asm.ADD, asm.SUB, asm.MUL, asm.DIV, asm.AND, asm.OR, asm.XOR, asm.SHL, asm.SHR, asm.LD8, asm.ST8, asm.LD32, asm.ST32, asm.JMP, asm.JEQ, asm.JNE, asm.JLT, asm.IO, asm.EVENT, asm.SLEEP
}

-- Pack the first four bytes of a Drift instruction in little-endian order.
bendjs (pack_word) do
  local function pack_word(op: uint32, a: uint32, b: uint32, c: uint32): uint32
    return op + 256_u32 * (a + 256_u32 * (b + 256_u32 * c))
  end
end

local function encode(index: uint32, a: uint32, b: uint32, c: uint32): uint32
    <cexport, codename 'encode'>
  if index >= #codes or a > 15 or b > 15 or c > 15 then return 0xffffffff_u32 end
  return pack_word(codes[index], a, b, c)
end

function word_to_u32(w) {
  let x = 0;
  for (let i = 0; w.$ === "WCon"; i++) {
    x |= Number(w.head) << i;
    w = w.tail;
  }
  return x >>> 0;
}

function u32_to_word(x) {
  let w = {$: "WNil"};
  for (let i = 31; i >= 0; i--) {
    w = {$: "WCon", head: ((x >>> i) & 1) === 1, tail: w};
  }
  return w;
}

function f32_show(x) {
  let s = "nan";
  for (let p = 1; x === x && p <= 9 && f32_round(s) !== x; p += 1) {
    s = String(Number(x.toExponential(p - 1)));
  }
  return (Object.is(x, -0) ? "-0" : s).replace("Infinity", "inf");
}

function f32_read(s) {
  const re = /^[\t\n\v\f\r ]*[+-]?((\d+(?:\.\d*)?|\.\d+)(e[+-]?\d+)?|inf(inity)?|nan)$/i;
  return re.test(s) ? {$: "Some", value: f32_round(
    s.replace(/inf\w*/i, "Infinity"))} : {$: "None"};
}

const f32_round = function f32_round(s) {
  const d = Number(s);
  const a = Math.abs(d);
  const f = Math.fround(a);
  const g = 2 * a - Math.min(f, 2 ** 128);
  if (g === f || Math.fround(g) !== g || g === Infinity) {
    return Math.sign(d) * f;
  }
  let k = 0;
  while (a * 2 ** k % 1 !== 0) {
    k += 1;
  }
  const [, i, r, e] = /(\d*)\.?(\d*)(?:e([+-]?\d+))?$/i.exec(s);
  const n = Number(e ?? 0) - r.length;
  const x = BigInt(i + r) * 2n ** BigInt(k) * 10n ** BigInt(Math.max(n, 0));
  const y = BigInt(a * 2 ** k) * 10n ** BigInt(Math.max(-n, 0));
  return Math.sign(d) * (x === y || x > y !== g > f ? f : g);
};

const char_new = function char_new(c) {
  return c < 55296 || c >= 57344 && c <= 1114111 ? String.fromCodePoint(c) : String.fromCharCode(56320, c >>> 22, c >>> 11 & 2047, c & 2047);
};

function char_len(s, at = 0) {
  return s[at] === "\uDC00" ? 4 : s.codePointAt(at) > 0xFFFF ? 2 : 1;
}

function char_code(s) {
  return s.length < 4 ? s.codePointAt(0)
    : (s.charCodeAt(1) << 22 | s.charCodeAt(2) << 11 | s.charCodeAt(3)) >>> 0;
}

function string_contains(s, p) {
  // A match starts at a Char, not inside a packed U32 code.
  for (let at = 0; at <= s.length; at += char_len(s, at)) {
    if (s.startsWith(p, at)) {
      return true;
    }
  }
  return false;
}

function str_in(s) {

function array_new(d, v, lgs) {
  if (d > 31 - lgs) {
    throw "bend: an array past the deepest block class 31 (depth + log2 words per cell)";
  }
  return Array(2 ** d).fill(v);
}

function array_node(a, b) {
  if (a.length !== b.length) {
    throw "bend: runtime fail-stop";
  }
  return a.concat(b);
}

function array_rmw(a, i, f) {
  const old = a[i %= a.length];
  a[i] = f(old);
  return tup_new(a, old);
}

// ===

function run_tail(f, x) {
  return {$: "$JMP", f: f.j?.f === f ? f.j : f, x};
}

function run_clo(j) {
  const f = (x) => run_loop(j(x));
  f.j = j;
  j.f = f;
  return f;
}

function run_loop(r) {
  while (r !== null && typeof r === "object" && r.$ === "$JMP") {
    r = r.f(r.x);
  }
  return r;
}

function run_lib(f, n) {
  return (...a) => a.length < n ? run_lib((...b) => f(...a, ...b), n - a.length)
    : run_loop(f(...a));
}

function $nelua_lower_function1(_nelua_lower_value2_0, _nelua_lower_value3_0, _nelua_lower_value4_0, _nelua_lower_value5_0) {
  const _x_0 = ((0 + 256) >>> 0);
  const _x_1 = (Math.imul(_x_0, _nelua_lower_value5_0) >>> 0);
  const _x_2 = ((0 + 256) >>> 0);
  const _x_3 = ((_nelua_lower_value4_0 + _x_1) >>> 0);
  const _x_4 = (Math.imul(_x_2, _x_3) >>> 0);
  const _x_5 = ((0 + 256) >>> 0);
  const _x_6 = ((_nelua_lower_value3_0 + _x_4) >>> 0);
  const _x_7 = (Math.imul(_x_5, _x_6) >>> 0);

}

nluint8_arr0_ptr mx_alloc(uintptr_t n) {
  uintptr_t size = n;
  if((size == 0)) {
    size = 1U;
  }
  size = ((size + 7) & 4294967288ULL);
  if((size > (262144U - melodica_core_st.arena_top))) {
    return ((nluint8_arr0_ptr)NULL);
  }
  uintptr_t start = melodica_core_st.arena_top;
  melodica_core_st.arena_top = (start + size);
  return ((nluint8_arr0_ptr)(&melodica_core_st.arena[start]));
}
